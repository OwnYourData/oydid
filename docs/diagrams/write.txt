@startuml
start
note
  Write DID
end note
partition Invocation {
  split
    :CLI;
  split again
    :Library;
  split again
    :API;
  end split
}
partition Preprocessing (requires private key) {
    note
        internal data format
        ----
        DID Identifier
        DID Document
        DID Keys
        DID Logs
    end note
  :generate base data;
}
partition Store Data (trustless, does not require private key) {
:validate input;
if (write method) then (local storage)
  :write to disc;
  note left
    files
    ----
     *.did - DID (full identifier)
     *.doc - DID Document
     *.log - DID Logs
  end note
else (online repository)
  partition Transaction {
    :write DID Document;
    :write DID Logs;
  }
endif
}
:Response;
stop
note right
  DID Identifier
end note
@enduml
